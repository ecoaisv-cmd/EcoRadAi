// ═══════════════════════════════════════════════════════════ // descargarPDFLite — EXCLUSIVA del informe lite // Captura el informe completo como una sola imagen y lo pagina // NO toca el informe médico ni la receta // ═══════════════════════════════════════════════════════════ async function descargarPDFLite() { const el = document.getElementById("report-lite"); if (!el) { showStatus("Genera primero el informe lite", "error"); return; } showStatus("Generando PDF lite...", "info"); // Clonar para no afectar el DOM original const clon = el.cloneNode(true); clon.style.cssText = "position:fixed;left:-99999px;top:0;width:800px;background:#fff;padding:0"; document.body.appendChild(clon); try { // Convertir Data URLs → Blob URLs (evita "Incomplete or corrupt PNG") const imagenes = clon.querySelectorAll('img'); for (const img of imagenes) { const src = img.getAttribute('src'); if (src && src.startsWith('data:image')) { try { const resp = await fetch(src); const blob = await resp.blob(); const blobUrl = URL.createObjectURL(blob); img.setAttribute('src', blobUrl); img.dataset.blobUrl = blobUrl; } catch (e) { console.warn('[PDF Lite] Imagen no convertida:', e); } } } await Promise.all(Array.from(imagenes).map(img => { if (img.complete) return Promise.resolve(); return new Promise(res => { img.onload = res; img.onerror = res; }); })); // Capturar el informe lite COMPLETO como una sola imagen const canvas = await html2canvas(clon, { scale: 2, useCORS: true, backgroundColor: "#ffffff", windowWidth: 800, scrollY: -window.scrollY }); const { jsPDF } = window.jspdf; const pdf = new jsPDF("p", "mm", "a4"); const pw = pdf.internal.pageSize.getWidth(); const ph = pdf.internal.pageSize.getHeight(); const mg = 10; const cw = pw - (mg * 2); const imgW = cw; const imgHTotal = (canvas.height * imgW) / canvas.width; if (imgHTotal <= ph - 2 * mg) { // Cabe en una sola página pdf.addImage(canvas.toDataURL("image/png"), "PNG", mg, mg, imgW, imgHTotal); } else { // Paginar en cortes limpios const pxPorMm = canvas.width / imgW; const altoPaginaPx = (ph - 2 * mg) * pxPorMm; let offsetY = 0; let pagina = 0; while (offsetY < canvas.height) { if (pagina > 0) pdf.addPage(); const altoCorte = Math.min(altoPaginaPx, canvas.height - offsetY); const temp = document.createElement("canvas"); temp.width = canvas.width; temp.height = altoCorte; const ctx = temp.getContext("2d"); ctx.fillStyle = "#ffffff"; ctx.fillRect(0, 0, temp.width, temp.height); ctx.drawImage(canvas, 0, offsetY, canvas.width, altoCorte, 0, 0, canvas.width, altoCorte); const altoMm = altoCorte / pxPorMm; pdf.addImage(temp.toDataURL("image/png"), "PNG", mg, mg, imgW, altoMm); offsetY += altoCorte; pagina++; } } // Pie de página const total = pdf.internal.getNumberOfPages(); for (let i = 1; i <= total; i++) { pdf.setPage(i); pdf.setFontSize(8); pdf.setTextColor(150); pdf.text("EcoRadAi · " + new Date().toLocaleDateString("es-ES"), mg, ph - 5); pdf.text("Página " + i + " de " + total, pw - mg, ph - 5, { align: "right" }); } const pacNom = (obtenerDatosPaciente().nombre || "paciente").replace(/\s+/g, "_"); pdf.save("Informe_Lite_" + pacNom + "_" + new Date().toISOString().slice(0, 10) + ".pdf"); showStatus("PDF lite descargado (" + total + " página" + (total > 1 ? "s" : "") + ")", "success"); } catch (e) { console.error("[descargarPDFLite]", e); showStatus("Error: " + e.message, "error"); } finally { clon.querySelectorAll('img[data-blob-url]').forEach(img => { if (img.dataset.blobUrl) URL.revokeObjectURL(img.dataset.blobUrl); }); document.body.removeChild(clon); } }
