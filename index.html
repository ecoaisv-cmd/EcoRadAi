const bloques = limpio.querySelectorAll(".report-section, .lv2-section, .lv2-semaforo, .lv2-firma"); for (const b of bloques) { const c = await html2canvas(b, { scale: 2, useCORS: true, backgroundColor: "#ffffff" }); const imgW = cw; const imgHTotal = (c.height * imgW) / c.width; // Si la sección cabe completa en la página restante, la agregamos tal cual if (y + imgHTotal <= ph - mg) { pdf.addImage(c.toDataURL("image/png"), "PNG", mg, y, imgW, imgHTotal); y += imgHTotal + 3; continue; } // Si NO cabe: la dividimos en porciones que sí quepan const altoDisponibleMm = ph - mg - y; if (altoDisponibleMm < 20) { // Poco espacio, saltamos de página pdf.addPage(); y = mg; addHeader(); } const pxPorMm = c.width / imgW; const altoPaginaPx = (ph - mg - y) * pxPorMm; let offsetY = 0; while (offsetY < c.height) { const altoCorte = Math.min(altoPaginaPx, c.height - offsetY); const temp = document.createElement("canvas"); temp.width = c.width; temp.height = altoCorte; const ctx = temp.getContext("2d"); ctx.drawImage(c, 0, offsetY, c.width, altoCorte, 0, 0, c.width, altoCorte); const altoMm = altoCorte / pxPorMm; pdf.addImage(temp.toDataURL("image/png"), "PNG", mg, y, imgW, altoMm); offsetY += altoCorte; y += altoMm + 2; if (offsetY < c.height) { pdf.addPage(); y = mg; addHeader(); } } }
